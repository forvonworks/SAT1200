\begin{textblock*}{158mm}(26mm,150.0mm)

\end{textblock*}
\begin{textblock*}{158mm}(26mm,162.5mm)

\begin{textblock*}{158mm}(26mm,170.5mm)
{\fontsize{13}{16}\selectfont axis $x=3$ and $f(-1)=f(7)$: the partner of $-1$ is $2(3)-(-1)=7$}

\cardbg{16}{33.0}{178}{43.0}
\pill{26}{36.0}{Per-period exponent}
\begin{textblock*}{158mm}(26mm,43.5mm)
{\fontsize{15}{18}\selectfont $a\,b^{t/n}$: multiply by $b$ every $n$ units of $t$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,52.5mm)
{\fontsize{13}{16}\selectfont doubling time $d$: $a\cdot2^{t/d}$ \qquad half-life $d$: $a\left(\tfrac12\right)^{t/d}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,64.0mm)
{\fontsize{13}{16}\selectfont $250(1.5)^{t/3}$: $+50\%$ every $3$ years (not every year)}
\end{textblock*}

\cardbg{16}{82.0}{178}{36.5}
\pill{26}{85.0}{Changing units (years to months)}
\begin{textblock*}{158mm}(26mm,94.0mm)
{\fontsize{15}{18}\selectfont put $\dfrac m{12}$ in place of $t$: \ $P_0b^t\ \Rightarrow\ P_0\,b^{m/12}$, monthly factor $b^{1/12}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,106.5mm)
{\fontsize{13}{16}\selectfont $1.2^{1/12}\approx1.0153$: about $1.5\%$ a month (not $20\div12$)}
\end{textblock*}

\cardbg{16}{124.5}{178}{36.5}
\pill{26}{127.5}{Rebuild $a\,b^x$ from two points}
\begin{textblock*}{158mm}(26mm,136.5mm)
{\fontsize{15}{18}\selectfont $\dfrac{y_2}{y_1}=b^{\,x_2-x_1}$ \qquad then $a=\dfrac{y_1}{b^{x_1}}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,149.0mm)
{\fontsize{13}{16}\selectfont $f(2)=18$, $f(5)=486$: $b^3=27$, $b=3$, $a=2$}
\end{textblock*}

\cardbg{16}{167.0}{178}{39.5}
\pill{26}{170.0}{Composition $f(g(x))$}
\begin{textblock*}{158mm}(26mm,177.5mm)
{\fontsize{15}{18}\selectfont inside first: put $g(x)$ where $f$ has $x$ \qquad $f(g(x))\neq g(f(x))$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,186.5mm)
{\fontsize{13}{16}\selectfont $f(f(x))$: use $f$ twice, starting from the inside}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,194.5mm)
{\fontsize{13}{16}\selectfont $f=2x+3$, $g=x^2-1$: \ $g(f(x))=(2x+3)^2-1=4x^2+12x+8$}
\end{textblock*}

\cardbg{16}{212.5}{178}{39.5}
\pill{26}{215.5}{Transformations $y=a\,f(x-h)+k$}
\begin{textblock*}{158mm}(26mm,223.0mm)
{\fontsize{15}{18}\selectfont right $h$, up $k$, stretch by $|a|$, flip if $a<0$ \qquad $f(x+3)$ moves LEFT}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,232.0mm)
{\fontsize{13}{16}\selectfont a point $(x,y)$ on $f$ goes to $(x+h,\ ay+k)$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,240.0mm)
{\fontsize{13}{16}\selectfont $f$ has vertex $(2,-3)$: \ $-2f(x+1)+5$ has vertex $(1,11)$}
\end{textblock*}
