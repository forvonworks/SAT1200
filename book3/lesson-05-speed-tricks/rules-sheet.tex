\begin{textblock*}{158mm}(26mm,96.0mm)
{\fontsize{15}{18}\selectfont $ax^2+bx+c=0$: \quad $r+s=-\dfrac ba$ \qquad $rs=\dfrac ca$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,108.5mm)
{\fontsize{13}{16}\selectfont $r^2+s^2=(r+s)^2-2rs$ \quad $\dfrac1r+\dfrac1s=\dfrac{r+s}{rs}$ \quad $|r-s|=\dfrac{\sqrt{D}}{|a|}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,120.5mm)
{\fontsize{13}{16}\selectfont $2x^2-7x+3=0$: \ $r+s=\tfrac72$, \ $rs=\tfrac32$, \ $r^2+s^2=\tfrac{49}4-3=\tfrac{37}4$}
\end{textblock*}

\end{textblock*}
\begin{textblock*}{158mm}(26mm,163.0mm)

\begin{textblock*}{158mm}(26mm,171.0mm)
{\fontsize{13}{16}\selectfont $y=x^2-5x+9$ and $y=2x+4$: \ $x^2-7x+5=0$, $x_1+x_2=7$, $y_1+y_2=2(7)+8=22$}

% ---- Add or subtract the equations (y=33.0 h=39.5 end 72.5) ----
\cardbg{16}{33.0}{178}{39.5}
\pill{26}{36.0}{Add or subtract the equations}
\begin{textblock*}{158mm}(26mm,43.5mm)

\begin{textblock*}{158mm}(26mm,60.5mm)

% ---- Factor and remainder theorem (y=78.5 h=39.5 end 118.0) ----
\cardbg{16}{78.5}{178}{39.5}
\pill{26}{81.5}{Factor and remainder theorem}
\begin{textblock*}{158mm}(26mm,89.0mm)
{\fontsize{15}{18}\selectfont remainder of $p(x)\div(x-a)$ is $p(a)$ \qquad $x-a$ is a factor $\Longleftrightarrow\ p(a)=0$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,98.0mm)
{\fontsize{13}{16}\selectfont $x^3+kx^2-5x+6$ has factor $x-2$: \ $8+4k-10+6=0\ \Rightarrow\ k=-1$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,106.0mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont Divide by $x+3$? Use $p(-3)$ \,--\, the sign flips.}
\end{textblock*}

% ---- Structure: identities (y=124.0 h=51.5 end 175.5) ----

\pill{26}{127.0}{Structure: identities}
\begin{textblock*}{158mm}(26mm,134.5mm)
{\fontsize{15}{18}\selectfont $a^2-b^2=(a-b)(a+b)$ \qquad $(a\pm b)^2=a^2\pm2ab+b^2$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,143.5mm)
{\fontsize{13}{16}\selectfont $a^2+b^2=(a+b)^2-2ab$ \quad $(a-b)^2=(a+b)^2-4ab$ \quad $x+\dfrac1x=k\Rightarrow x^2+\dfrac1{x^2}=k^2-2$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,155.5mm)
{\fontsize{13}{16}\selectfont $x+y=9$, $xy=14$: \ $x^2+y^2=81-28=53$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,163.5mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont Spot a perfect square inside: $x^2-y^2-6y-9=x^2-(y+3)^2$.}
\end{textblock*}

% ---- Factor, never divide by an unknown (y=181.5 h=39.5 end 221.0) ----
\cardbg{16}{181.5}{178}{39.5}
\pill{26}{184.5}{Factor, never divide by an unknown}
\begin{textblock*}{158mm}(26mm,192.0mm)
{\fontsize{15}{18}\selectfont move everything to one side $\Rightarrow$ factor $\Rightarrow$ zero-product rule}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,201.0mm)
{\fontsize{13}{16}\selectfont $(x-7)(x+3)=5(x-7)\ \Rightarrow\ (x-7)(x-2)=0\ \Rightarrow\ x=7$ or $x=2$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,209.0mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont Dividing by $x-7$ would lose the root $x=7$.}
\end{textblock*}

% ---- Percent multipliers (y=227.0 h=48.0 end 275.0) ----
\cardbg{16}{227.0}{178}{48.0}
\pill{26}{230.0}{Percent multipliers}
\begin{textblock*}{158mm}(26mm,237.5mm)
{\fontsize{15}{18}\selectfont $+r\%\Rightarrow\times\left(1+\dfrac r{100}\right)$ \qquad $-r\%\Rightarrow\times\left(1-\dfrac r{100}\right)$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,250.0mm)
{\fontsize{13}{16}\selectfont successive changes multiply; \ undo a change by dividing: \ old $=\dfrac{\text{new}}{\text{multiplier}}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,262.0mm)
{\fontsize{13}{16}\selectfont $+20\%$ then $-20\%$: \ $1.2\times0.8=0.96$, a $4\%$ drop (not $0\%$)}
\end{textblock*}
