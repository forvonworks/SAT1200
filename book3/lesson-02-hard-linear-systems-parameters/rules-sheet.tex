\begin{textblock*}{158mm}(26mm,52.0mm)

\end{textblock*}
\begin{textblock*}{158mm}(26mm,65.5mm)
{\fontsize{13}{16}\selectfont $y$-intercept $\dfrac CB$ \qquad $x$-intercept $\dfrac CA$ \qquad point-slope: $y-y_1=m(x-x_1)$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,77.5mm)
{\fontsize{13}{16}\selectfont $3x+4y=12$: \ slope $-\tfrac34$, \ $y$-intercept $3$, \ $x$-intercept $4$}
\end{textblock*}
% ---- Slope, parallel, perpendicular (y=95.5 h=44.5 end 140.0)
\cardbg{16}{95.5}{178}{44.5}
\pill{26}{98.5}{Slope, parallel, perpendicular}
\begin{textblock*}{158mm}(26mm,107.5mm)
{\fontsize{15}{18}\selectfont $m=\dfrac{y_2-y_1}{x_2-x_1}$ \qquad parallel: $m_1=m_2$ \qquad perpendicular: $m_1m_2=-1$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,121.0mm)
{\fontsize{13}{16}\selectfont slope $-\tfrac23$ $\Rightarrow$ perpendicular slope $\tfrac32$ \ (flip the fraction, change the sign)}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,129.0mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont Standard form: $Ax+By=C$ and $Ax+By=D$ are parallel; $Bx-Ay=D$ is perpendicular to the first.}
\end{textblock*}
% ---- Number of solutions: the $a:b:c$ ratio test (y=146.0 h=60.5 end 206.5)
\cardbg{16}{146.0}{178}{60.5}
\pill{26}{149.0}{Number of solutions: the $a:b:c$ ratio test}
\begin{textblock*}{158mm}(26mm,158.0mm)
{\fontsize{15}{18}\selectfont $a_1x+b_1y=c_1$ and $a_2x+b_2y=c_2$: \ $\dfrac{a_1}{a_2}\ne\dfrac{b_1}{b_2}\Rightarrow$ exactly one}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,171.5mm)
{\fontsize{13}{16}\selectfont no solution: $\dfrac{a_1}{a_2}=\dfrac{b_1}{b_2}\ne\dfrac{c_1}{c_2}$ \qquad infinitely many: all three ratios equal}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,183.5mm)
{\fontsize{13}{16}\selectfont $kx+12y=9$ and $5x+4y=7$: \ $\dfrac k5=\dfrac{12}4\Rightarrow k=15$, \ and $9\ne3\cdot7$, so no solution}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,195.5mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont Slope form: same slope and different intercept = no solution; the same line = infinitely many.}
\end{textblock*}
% ---- A parameter in a system (y=212.5 h=42.5 end 255.0)
\cardbg{16}{212.5}{178}{42.5}
\pill{26}{215.5}{A parameter in a system}
\begin{textblock*}{158mm}(26mm,223.0mm)
{\fontsize{15}{18}\selectfont ratios equal $\Rightarrow$ solve for $k$ $\Rightarrow$ \textbf{check the constants} for each $k$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,232.0mm)
{\fontsize{13}{16}\selectfont $\dfrac k4=\dfrac9k\Rightarrow k=\pm6$: \ one value gives infinitely many solutions, the other none}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,244.0mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont \textit{Which could be the second equation?} Same $x,y$-ratio as the given one, different constant.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,52.5mm)

% ---- One variable: identity or contradiction (y=81.5 h=46.5 end 128.0)
\cardbg{16}{81.5}{178}{46.5}
\pill{26}{84.5}{One variable: identity or contradiction}
\begin{textblock*}{158mm}(26mm,92.0mm)
{\fontsize{15}{18}\selectfont $ax+b=cx+d$: \ $a\ne c$ \ $\Rightarrow$ \ exactly one solution}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,101.0mm)
{\fontsize{13}{16}\selectfont $a=c$ and $b=d$: infinitely many \qquad $a=c$ and $b\ne d$: no solution}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,109.0mm)
{\fontsize{13}{16}\selectfont $3(2x-k)+x=7x+12\Rightarrow 7x-3k=7x+12$, so $k=-4$ gives infinitely many}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,117.0mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont Variables cancel and a true statement is left: infinitely many; a false statement: no solution.}
\end{textblock*}
% ---- Interpreting a linear model (y=134.0 h=43.5 end 177.5)
\cardbg{16}{134.0}{178}{43.5}
\pill{26}{137.0}{Interpreting a linear model}
\begin{textblock*}{158mm}(26mm,144.5mm)
{\fontsize{15}{18}\selectfont $y=mx+b$: \ $m$ = change in $y$ per 1 unit of $x$ \quad $b$ = value at $x=0$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,153.5mm)
{\fontsize{13}{16}\selectfont $Ax+By=C$: \ $x$-intercept $\dfrac CA$ (all of one item), \ each $+1$ in $x$ changes $y$ by $-\dfrac AB$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,165.5mm)
{\fontsize{13}{16}\selectfont $3x+5y=600$: \ the point $(0,120)$ means 0 of the first item and 120 of the second}
\end{textblock*}
% ---- Linear function from two points or a table (y=183.5 h=44.5 end 228.0)
\cardbg{16}{183.5}{178}{44.5}
\pill{26}{186.5}{Linear function from two points or a table}
\begin{textblock*}{158mm}(26mm,195.5mm)
{\fontsize{15}{18}\selectfont $m=\dfrac{\Delta y}{\Delta x}$ \qquad $f(x)=f(x_1)+m(x-x_1)$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,209.0mm)
{\fontsize{13}{16}\selectfont equal steps in $x$ give equal changes in $y$: \ $f(a+3)-f(a)=12\Rightarrow f(a+10)-f(a)=40$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,217.0mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont Shift right $h$: $f(x-h)$. Shift up $k$: $f(x)+k$. Reading a table: compute $\Delta y\div\Delta x$ first.}
\end{textblock*}
% ---- Means and totals (y=234.0 h=44.5 end 278.5)
\cardbg{16}{234.0}{178}{44.5}
\pill{26}{237.0}{Means and totals}
\begin{textblock*}{158mm}(26mm,246.0mm)
{\fontsize{15}{18}\selectfont mean $=\dfrac{\text{total}}{\text{count}}$ \qquad combined mean $=\dfrac{n_1\bar x_1+n_2\bar x_2}{n_1+n_2}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,259.5mm)
{\fontsize{13}{16}\selectfont missing group: \ $24(78)+36m=60(84)\Rightarrow m=88$ \quad (work with totals)}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,267.5mm)
{\itshape\color{black!70}\fontsize{11}{14}\selectfont Two means can be averaged directly only when the groups have the same size.}
\end{textblock*}
