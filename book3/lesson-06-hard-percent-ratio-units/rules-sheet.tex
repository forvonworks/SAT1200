% ---- Percent as a multiplier (y=40.0 h=34.5 end 74.5)
\cardbg{16}{40.0}{178}{34.5}
\pill{26}{43.0}{Percent as a multiplier}
\begin{textblock*}{158mm}(26mm,51.5mm)
{\fontsize{15}{18}\selectfont $p\%$ of $N=\dfrac p{100}\,N$ \qquad new $=N\left(1\pm\dfrac p{100}\right)$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,64.0mm)
{\fontsize{13}{16}\selectfont $+15\%\Rightarrow\times1.15$ \qquad $-15\%\Rightarrow\times0.85$ \qquad $+150\%\Rightarrow\times2.5$}
\end{textblock*}
% ---- Percent change (y=80.5 h=37.0 end 117.5)
\cardbg{16}{80.5}{178}{37.0}
\pill{26}{83.5}{Percent change}
\begin{textblock*}{158mm}(26mm,92.0mm)
{\fontsize{15}{18}\selectfont $\%\text{ change}=\dfrac{\text{new}-\text{old}}{\text{old}}\times100\%$ \quad (the base is the \textbf{old} value)}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,104.5mm)
{\fontsize{13}{16}\selectfont $40\to46$: \ $\dfrac{6}{40}=15\%$ \qquad $12\to9$: \ $-\dfrac{3}{12}=-25\%$}
\end{textblock*}
% ---- Percent change vs percentage points (y=123.5 h=34.5 end 158.0)
\cardbg{16}{123.5}{178}{34.5}
\pill{26}{126.5}{Percent change vs percentage points}
\begin{textblock*}{158mm}(26mm,135.0mm)
{\fontsize{15}{18}\selectfont points $=$ rate$_2-$rate$_1$ \qquad percent change $=\dfrac{\text{points}}{\text{rate}_1}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,147.5mm)
{\fontsize{13}{16}\selectfont $40\%\to46\%$: \ $+6$ percentage points, which is a $+15\%$ change}
\end{textblock*}
% ---- Successive changes: multiply, never add (y=164.0 h=34.5 end 198.5)
\cardbg{16}{164.0}{178}{34.5}
\pill{26}{167.0}{Successive changes: multiply, never add}
\begin{textblock*}{158mm}(26mm,175.5mm)
{\fontsize{15}{18}\selectfont factor $=f_1\cdot f_2\cdots$ \qquad up $x\%$ then down $x\%$: $1-\left(\dfrac{x}{100}\right)^2$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,188.0mm)
{\fontsize{13}{16}\selectfont $+20\%$ then $-20\%$: \ $1.2\times0.8=0.96$, a net $-4\%$ (not $0$)}
\end{textblock*}
% ---- Reverse percent: divide by the factor (y=204.5 h=37.0 end 241.5)
\cardbg{16}{204.5}{178}{37.0}
\pill{26}{207.5}{Reverse percent: divide by the factor}
\begin{textblock*}{158mm}(26mm,216.0mm)
{\fontsize{15}{18}\selectfont original $=\dfrac{\text{final}}{\text{factor}}$ \qquad to undo $\times f$ multiply by $\dfrac1f$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,228.5mm)
{\fontsize{13}{16}\selectfont final \$102 after $-15\%$: \ $102\div0.85=120$ \qquad undo $+60\%$: \ $1-\dfrac1{1.6}=37.5\%$ off}
\end{textblock*}
% ---- ``$n\%$ more'' vs ``$n$ times as much'' (y=247.5 h=34.5 end 282.0)
\cardbg{16}{247.5}{178}{34.5}
\pill{26}{250.5}{``$n\%$ more'' vs ``$n$ times as much''}
\begin{textblock*}{158mm}(26mm,259.0mm)
{\fontsize{15}{18}\selectfont $n\%$ more $=\times\left(1+\dfrac n{100}\right)$ \qquad $n$ times as much $=\times n$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,271.5mm)
{\fontsize{13}{16}\selectfont $150\%$ more $=2.5$ times \qquad $150\%$ of $=1.5$ times \qquad $40\%$ less $=\times0.6$}
\end{textblock*}

% ---- Simple vs compound interest (y=76.0 h=34.5 end 110.5)
\cardbg{16}{76.0}{178}{34.5}
\pill{26}{79.0}{Simple vs compound interest}
\begin{textblock*}{158mm}(26mm,87.5mm)
{\fontsize{15}{18}\selectfont simple: $A=P(1+rt)$ \qquad compound: $A=P(1+r)^t$ \qquad $P\left(1+\dfrac rm\right)^{mt}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,100.0mm)
{\fontsize{13}{16}\selectfont $12\%$ compounded monthly: \ $P(1.01)^{12t}$ \qquad $1.5\%$ a month $\Rightarrow1.015^{12}-1\approx19.6\%$ a year}
\end{textblock*}
% ---- Mixtures and concentration (y=116.5 h=30.0 end 146.5)
\cardbg{16}{116.5}{178}{30.0}
\pill{26}{119.5}{Mixtures and concentration}
\begin{textblock*}{158mm}(26mm,127.5mm)
{\fontsize{15}{18}\selectfont amount $=$ concentration $\times$ volume \qquad amount in $=$ amount out}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,136.0mm)
{\fontsize{13}{16}\selectfont $0.40(6)+0.10x=0.30(6+x)\Rightarrow x=3$ \qquad water or evaporation: \textbf{solute stays fixed}}
\end{textblock*}
% ---- Weighted average and composition (y=152.5 h=37.0 end 189.5)
\cardbg{16}{152.5}{178}{37.0}
\pill{26}{155.5}{Weighted average and composition}
\begin{textblock*}{158mm}(26mm,164.0mm)
{\fontsize{15}{18}\selectfont mean $=\dfrac{\sum n_ix_i}{\sum n_i}$ \qquad fraction of group 2: $f=\dfrac{\text{mean}-x_1}{x_2-x_1}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,176.5mm)
{\fontsize{13}{16}\selectfont $20\%$ and $50\%$ mixed to $38\%$: \ $f=\dfrac{18}{30}=0.6$ \qquad table: pick the \textbf{row or column} as base}
\end{textblock*}
% ---- Ratios: part-to-part vs part-to-whole (y=195.5 h=37.0 end 232.5)
\cardbg{16}{195.5}{178}{37.0}
\pill{26}{198.5}{Ratios: part-to-part vs part-to-whole}
\begin{textblock*}{158mm}(26mm,207.0mm)
{\fontsize{15}{18}\selectfont $a:b\Rightarrow\dfrac a{a+b}$ of the whole \qquad write $a=ak$, $b=bk$ and solve}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,219.5mm)
{\fontsize{13}{16}\selectfont $3:5$ of $96$: $36$ and $60$ \qquad $3k:7k$, remove $8$ red $\to3:5$: \ $\dfrac{3k}{7k-8}=\dfrac35$}
\end{textblock*}
% ---- Percent error and relative change (y=238.5 h=37.0 end 275.5)

\pill{26}{241.5}{Percent error and relative change}
\begin{textblock*}{158mm}(26mm,250.0mm)
{\fontsize{15}{18}\selectfont percent error $=\dfrac{|\text{measured}-\text{true}|}{\text{true}}\times100\%$ \qquad base $=$ true value}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,262.5mm)
{\fontsize{13}{16}\selectfont radius $+5\%\Rightarrow$ area $\times1.05^2=1.1025$, so $+10.25\%$ \qquad measured $4\%$ low: true $=\dfrac m{0.96}$}
\end{textblock*}
