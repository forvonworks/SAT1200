% ---- Percent as a multiplier (y=40.0 h=37.5 end 77.5)
\cardbg{16}{40.0}{178}{37.5}
\pill{26}{43.0}{Percent as a multiplier}
\begin{textblock*}{158mm}(26mm,52.0mm)
{\fontsize{15}{18}\selectfont $p\%$ of $N=\dfrac p{100}\,N$ \qquad new $=N\left(1\pm\dfrac p{100}\right)$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,65.5mm)
{\fontsize{13}{16}\selectfont $+15\%\Rightarrow\times1.15$ \qquad $-15\%\Rightarrow\times0.85$ \qquad $+150\%\Rightarrow\times2.5$}
\end{textblock*}
% ---- Percent change (y=83.5 h=40.5 end 124.0)
\cardbg{16}{83.5}{178}{40.5}
\pill{26}{86.5}{Percent change}
\begin{textblock*}{158mm}(26mm,95.5mm)
{\fontsize{15}{18}\selectfont $\%\text{ change}=\dfrac{\text{new}-\text{old}}{\text{old}}\times100\%$ \quad (the base is the \textbf{old} value)}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,109.0mm)
{\fontsize{13}{16}\selectfont $40\to46$: \ $\dfrac{6}{40}=15\%$ \qquad $12\to9$: \ $-\dfrac{3}{12}=-25\%$}
\end{textblock*}
% ---- Percent change vs percentage points (y=130.0 h=37.5 end 167.5)
\cardbg{16}{130.0}{178}{37.5}
\pill{26}{133.0}{Percent change vs percentage points}
\begin{textblock*}{158mm}(26mm,142.0mm)
{\fontsize{15}{18}\selectfont points $=$ rate$_2-$rate$_1$ \qquad percent change $=\dfrac{\text{points}}{\text{rate}_1}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,155.5mm)
{\fontsize{13}{16}\selectfont $40\%\to46\%$: \ $+6$ percentage points, which is a $+15\%$ change}
\end{textblock*}
% ---- Successive changes: multiply, never add (y=173.5 h=37.5 end 211.0)
\cardbg{16}{173.5}{178}{37.5}
\pill{26}{176.5}{Successive changes: multiply, never add}
\begin{textblock*}{158mm}(26mm,185.5mm)
{\fontsize{15}{18}\selectfont factor $=f_1\cdot f_2\cdots$ \qquad up $x\%$ then down $x\%$: $1-\left(\dfrac{x}{100}\right)^2$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,199.0mm)
{\fontsize{13}{16}\selectfont $+20\%$ then $-20\%$: \ $1.2\times0.8=0.96$, a net $-4\%$ (not $0$)}
\end{textblock*}
% ---- Reverse percent: divide by the factor (y=217.0 h=40.5 end 257.5)
\cardbg{16}{217.0}{178}{40.5}
\pill{26}{220.0}{Reverse percent: divide by the factor}
\begin{textblock*}{158mm}(26mm,229.0mm)
{\fontsize{15}{18}\selectfont original $=\dfrac{\text{final}}{\text{factor}}$ \qquad to undo $\times f$ multiply by $\dfrac1f$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,242.5mm)
{\fontsize{13}{16}\selectfont final \$102 after $-15\%$: \ $102\div0.85=120$ \qquad undo $+60\%$: \ $1-\dfrac1{1.6}=37.5\%$ off}
\end{textblock*}
% ---- ``$n\%$ more'' vs ``$n$ times as much'' (y=263.5 h=37.5 end 301.0)
\cardbg{16}{263.5}{178}{37.5}
\pill{26}{266.5}{``$n\%$ more'' vs ``$n$ times as much''}
\begin{textblock*}{158mm}(26mm,275.5mm)
{\fontsize{15}{18}\selectfont $n\%$ more $=\times\left(1+\dfrac n{100}\right)$ \qquad $n$ times as much $=\times n$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,289.0mm)
{\fontsize{13}{16}\selectfont $150\%$ more $=2.5$ times \qquad $150\%$ of $=1.5$ times \qquad $40\%$ less $=\times0.6$}
\end{textblock*}

% ---- Simple vs compound interest (y=79.5 h=37.5 end 117.0)
\cardbg{16}{79.5}{178}{37.5}
\pill{26}{82.5}{Simple vs compound interest}
\begin{textblock*}{158mm}(26mm,91.5mm)
{\fontsize{15}{18}\selectfont simple: $A=P(1+rt)$ \qquad compound: $A=P(1+r)^t$ \qquad $P\left(1+\dfrac rm\right)^{mt}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,105.0mm)
{\fontsize{13}{16}\selectfont $12\%$ compounded monthly: \ $P(1.01)^{12t}$ \qquad $1.5\%$ a month $\Rightarrow1.015^{12}-1\approx19.6\%$ a year}
\end{textblock*}
% ---- Mixtures and concentration (y=123.0 h=31.5 end 154.5)
\cardbg{16}{123.0}{178}{31.5}
\pill{26}{126.0}{Mixtures and concentration}
\begin{textblock*}{158mm}(26mm,133.5mm)
{\fontsize{15}{18}\selectfont amount $=$ concentration $\times$ volume \qquad amount in $=$ amount out}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,142.5mm)
{\fontsize{13}{16}\selectfont $0.40(6)+0.10x=0.30(6+x)\Rightarrow x=3$ \qquad water or evaporation: \textbf{solute stays fixed}}
\end{textblock*}
% ---- Weighted average and composition (y=160.5 h=40.5 end 201.0)
\cardbg{16}{160.5}{178}{40.5}
\pill{26}{163.5}{Weighted average and composition}
\begin{textblock*}{158mm}(26mm,172.5mm)
{\fontsize{15}{18}\selectfont mean $=\dfrac{\sum n_ix_i}{\sum n_i}$ \qquad fraction of group 2: $f=\dfrac{\text{mean}-x_1}{x_2-x_1}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,186.0mm)
{\fontsize{13}{16}\selectfont $20\%$ and $50\%$ mixed to $38\%$: \ $f=\dfrac{18}{30}=0.6$ \qquad table: pick the \textbf{row or column} as base}
\end{textblock*}
% ---- Ratios: part-to-part vs part-to-whole (y=207.0 h=40.5 end 247.5)
\cardbg{16}{207.0}{178}{40.5}
\pill{26}{210.0}{Ratios: part-to-part vs part-to-whole}
\begin{textblock*}{158mm}(26mm,219.0mm)
{\fontsize{15}{18}\selectfont $a:b\Rightarrow\dfrac a{a+b}$ of the whole \qquad write $a=ak$, $b=bk$ and solve}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,232.5mm)
{\fontsize{13}{16}\selectfont $3:5$ of $96$: $36$ and $60$ \qquad $3k:7k$, remove $8$ red $\to3:5$: \ $\dfrac{3k}{7k-8}=\dfrac35$}
\end{textblock*}
% ---- Rates, work and flow (y=253.5 h=40.5 end 294.0)
\cardbg{16}{253.5}{178}{40.5}
\pill{26}{256.5}{Rates, work and flow}
\begin{textblock*}{158mm}(26mm,265.5mm)
{\fontsize{15}{18}\selectfont distance $=$ rate $\times$ time \qquad work together: \textbf{add rates} $\dfrac1a+\dfrac1b$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,279.0mm)
{\fontsize{13}{16}\selectfont fills in 6 h and 12 h, drains in 9 h: \ $\dfrac16+\dfrac1{12}-\dfrac19=\dfrac5{36}$, so $7.2$ h}
\end{textblock*}

\pill{26}{82.5}{Unit conversion chains}
\begin{textblock*}{158mm}(26mm,90.0mm)
{\fontsize{15}{18}\selectfont multiply by fractions equal to $1$ \qquad squared units: \textbf{square} the factor, cubed: \textbf{cube}}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,99.0mm)
{\fontsize{13}{16}\selectfont $1\text{ m}^2=100^2=10^4\text{ cm}^2$ \qquad $1\text{ m}^3=10^6\text{ cm}^3=1000\text{ L}$}
\end{textblock*}
% ---- Similar figures: scale factor $k$ (y=117.0 h=40.5 end 157.5)

\pill{26}{120.0}{Similar figures: scale factor $k$}
\begin{textblock*}{158mm}(26mm,129.0mm)
{\fontsize{15}{18}\selectfont length $\times k$ \qquad area $\times k^2$ \qquad volume $\times k^3$ \qquad $k=\sqrt{\dfrac{A_2}{A_1}}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,142.5mm)
{\fontsize{13}{16}\selectfont $k=3$: area $\times9$, volume $\times27$ \qquad areas $20$ and $45$: \ $k=\dfrac32$}
\end{textblock*}
% ---- Scale drawings and density (y=163.5 h=37.5 end 201.0)

\pill{26}{166.5}{Scale drawings and density}
\begin{textblock*}{158mm}(26mm,175.5mm)
{\fontsize{15}{18}\selectfont map: $1$ cm $=d$ km $\Rightarrow1\text{ cm}^2=d^2\text{ km}^2$ \qquad density $=\dfrac{\text{mass}}{\text{volume}}$}
\end{textblock*}
\begin{textblock*}{158mm}(26mm,189.0mm)
{\fontsize{13}{16}\selectfont same material: mass $\propto$ volume $\propto k^3$ \qquad $2.7\text{ g/cm}^3=2700\text{ kg/m}^3$}
\end{textblock*}
% ---- Percent error and relative change (y=207.0 h=40.5 end 247.5)

\pill{26}{210.0}{Percent error and relative change}
\begin{textblock*}{158mm}(26mm,219.0mm)

\end{textblock*}
\begin{textblock*}{158mm}(26mm,232.5mm)
{\fontsize{13}{16}\selectfont radius $+5\%\Rightarrow$ area $\times1.05^2=1.1025$, so $+10.25\%$ \qquad measured $4\%$ low: true $=\dfrac m{0.96}$}
