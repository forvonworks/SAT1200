% ---- CARD 2: Factoring when a=1 ----
\cardbg{16}{96}{178}{55}
\pill{26}{99}{Factoring when $a=1$}

\begin{textblock*}{158mm}(26mm,112mm)
{\fontsize{11}{14}\selectfont\color{ink}
Find two numbers that multiply to $c$ and add to $b$.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,124mm)
{\fontsize{13}{16}\selectfont
$x^2+5x+6=0\ \Longrightarrow\ (x+2)(x+3)=0\ \Longrightarrow\ x=-2$ or $-3$}
\end{textblock*}

% ---- CARD 3: GCF first ----
\cardbg{16}{159}{178}{55}
\pill{26}{162}{Factor out the GCF first}

\begin{textblock*}{158mm}(26mm,175mm)
{\fontsize{11}{14}\selectfont\color{ink}
If every term shares a common factor, pull it out before factoring the rest.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,187mm)
{\fontsize{13}{16}\selectfont
$2x^2+8x+6=0\ \Longrightarrow\ 2(x+1)(x+3)=0\ \Longrightarrow\ x=-1$ or $-3$}
\end{textblock*}

% ---- CARD 4: Zero-product property ----
\cardbg{16}{33}{178}{90}
\pill{26}{36}{Zero-product property}

\begin{textblock*}{158mm}(26mm,49mm)
{\fontsize{11}{14}\selectfont\color{ink}
Works only when the product equals exactly $0$. If it doesn't, rearrange
everything to one side $=0$ first.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,64mm)
{\fontsize{12}{15}\selectfont
$(x-3)(x+1)=5$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,75mm)
{\fontsize{12}{15}\selectfont
$x^2-2x-3=5\ \Longrightarrow\ x^2-2x-8=0\ \Longrightarrow\ (x-4)(x+2)=0$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,90mm)
{\fontsize{13}{16}\selectfont\color{accent}
$x=4$ or $x=-2$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,105mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
A question often asks for the sum or product of the roots, not the roots
themselves -- e.g. sum $=4+(-2)=2$.}
\end{textblock*}
