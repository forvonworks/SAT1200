\cardbg{16}{40}{178}{95}
\pill{26}{43}{Area formulas (maydon)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{10.5}{14}\selectfont
\begin{tabular}{|p{48mm}|p{48mm}|p{49mm}|}
\hline
\rowcolor{accent}
{\color{white}\textbf{Shape}} & {\color{white}\textbf{Formula}} & {\color{white}\textbf{Variables}} \\ \hline
Rectangle & $A=lw$ & $l$=length, $w$=width \\ \hline
Triangle & $A=\dfrac12bh$ & $b$=base, $h$=height \\ \hline
Parallelogram & $A=bh$ & $b$=base, $h$=height \\ \hline
Trapezoid & $A=\dfrac12(a+b)h$ & $a,b$=parallel sides \\ \hline
Circle & $A=\pi r^2$ & $r$=radius \\ \hline
\end{tabular}}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,122mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
If given the \textbf{diameter}, halve it first: $r=d/2$. $d=18\ \Rightarrow\ r=9\ \Rightarrow\ A=81\pi$, not $324\pi$.}
\end{textblock*}

% ---- CARD 2: Volume formulas ----
\cardbg{16}{33}{178}{98}
\pill{26}{36}{Volume formulas (hajm)}

\begin{textblock*}{158mm}(26mm,49mm)
{\fontsize{10.5}{14}\selectfont
\begin{tabular}{|p{45mm}|p{45mm}|p{55mm}|}
\hline
\rowcolor{accent}
{\color{white}\textbf{Solid}} & {\color{white}\textbf{Formula}} & {\color{white}\textbf{Note}} \\ \hline
Rectangular prism & $V=lwh$ & product of 3 dimensions \\ \hline
Cylinder & $V=\pi r^2h$ & base area $\times$ height \\ \hline
Cone & $V=\dfrac13\pi r^2h$ & one-third of same-$r,h$ cylinder \\ \hline
Sphere & $V=\dfrac43\pi r^3$ & depends only on $r$ -- no height \\ \hline
\end{tabular}}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,113mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
Never drop the $\frac13$ (cone) or $\frac43$ (sphere) -- a cone is exactly
one-third of the matching cylinder's volume.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,124mm)
{\fontsize{12}{15}\selectfont
$r=6$: $V=\dfrac43\pi(6)^3=\dfrac43\pi(216)=288\pi$}
\end{textblock*}

% ---- CARD 3: Composite figures ----
\cardbg{16}{141}{178}{80}
\pill{26}{144}{Composite figures: add or subtract}

\begin{textblock*}{158mm}(26mm,157mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
Pieces joined together $\to$ \textbf{add}. A piece cut out/a hole $\to$
\textbf{subtract}.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,169mm)
{\fontsize{11}{14}\selectfont
Joined: $10\times6$ rect. $+\,4\times3$ rect. $=60+12=72$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,181mm)
{\fontsize{11}{14}\selectfont
Hole: $10\times10$ square, radius-$3$ hole $=100-9\pi$ (not $100+9\pi$)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,196mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Sanity check: subtracting a piece must give a result smaller than the
whole, never larger.}
\end{textblock*}
