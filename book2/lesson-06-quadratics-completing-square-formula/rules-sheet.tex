% ---- Completing the square (y=81 h=40, end 121) ----
\cardbg{16}{81}{178}{40}
\pill{26}{84}{Completing the square}

\begin{textblock*}{158mm}(26mm,94.0mm)
{\fontsize{15}{18}\selectfont $x^2+bx\ \Rightarrow\ \left(x+\dfrac{b}{2}\right)^2-\left(\dfrac{b}{2}\right)^2$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,108.5mm)
{\fontsize{13}{16}\selectfont $x^2-6x+5=(x-3)^2-9+5=(x-3)^2-4$}
\end{textblock*}

% ---- Completing the square when $a\neq1$ (y=128 h=34, end 162) ----
\cardbg{16}{128}{178}{34}
\pill{26}{131}{Completing the square when $a\neq1$}

\begin{textblock*}{158mm}(26mm,141.0mm)
{\fontsize{13}{15}\selectfont factor $a$ out of the $x^2$ and $x$ terms first}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,149.5mm)
{\fontsize{13}{16}\selectfont $2x^2+8x+5=2(x^2+4x)+5=2(x+2)^2-3$}
\end{textblock*}

% ---- Vertex from standard form (y=169 h=40, end 209) ----
\cardbg{16}{169}{178}{40}
\pill{26}{172}{Vertex from standard form}

\begin{textblock*}{158mm}(26mm,182.0mm)
{\fontsize{15}{18}\selectfont $y=ax^2+bx+c$:\quad $h=-\dfrac{b}{2a}$,\quad $k=f(h)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,196.5mm)
{\fontsize{13}{16}\selectfont $y=2x^2-8x+3$: \ $h=2$, \ $k=f(2)=-5$, \ vertex $(2,-5)$}
\end{textblock*}

% ---- Maximum or minimum value (y=216 h=34, end 250) ----
\cardbg{16}{216}{178}{34}
\pill{26}{219}{Maximum or minimum value}

\begin{textblock*}{158mm}(26mm,229.0mm)
{\fontsize{13}{15}\selectfont the extreme value is $k$:\quad $a>0\Rightarrow$ min,\quad $a<0\Rightarrow$ max}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,237.5mm)
{\fontsize{13}{16}\selectfont $y=-x^2+4x+1=-(x-2)^2+5$: \ max $=5$ at $x=2$}
\end{textblock*}

% ---- Square-root method (y=33 h=34, end 67) ----
\cardbg{16}{33}{178}{34}
\pill{26}{36}{Square-root method}

\begin{textblock*}{158mm}(26mm,46.0mm)
{\fontsize{15}{18}\selectfont $(x-h)^2=k\ \Rightarrow\ x=h\pm\sqrt{k}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,54.5mm)
{\fontsize{13}{16}\selectfont $(x-3)^2=7\ \Rightarrow\ x=3\pm\sqrt7$}
\end{textblock*}

% ---- Quadratic formula (y=74 h=46, end 120) ----
\cardbg{16}{74}{178}{46}
\pill{26}{77}{Quadratic formula}

\begin{textblock*}{158mm}(26mm,87.0mm)
{\fontsize{17}{20}\selectfont $ax^2+bx+c=0\ \Rightarrow\ x=\dfrac{-b\pm\sqrt{b^2-4ac}}{2a}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,102.5mm)
{\fontsize{13}{16}\selectfont $x^2+4x+1=0$: \ $a=1,\ b=4,\ c=1\ \Rightarrow\ x=\dfrac{-4\pm\sqrt{12}}{2}$}
\end{textblock*}

% ---- Sign care when $b<0$ (y=127 h=38, end 165) ----
\cardbg{16}{127}{178}{38}
\pill{26}{130}{Sign care when $b<0$}

\begin{textblock*}{158mm}(26mm,140.0mm)
{\fontsize{13}{15}\selectfont $-b$ is positive when $b$ is negative}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,148.5mm)
{\fontsize{13}{16}\selectfont $x^2-6x+4=0$: \ $b=-6$, \ $-b=6$ \ $\Rightarrow\ x=\dfrac{6\pm\sqrt{36-16}}{2}=\dfrac{6\pm\sqrt{20}}{2}$}
\end{textblock*}

% ---- Simplify the radical answer (y=172 h=46, end 218) ----
\cardbg{16}{172}{178}{46}
\pill{26}{175}{Simplify the radical answer}

\begin{textblock*}{158mm}(26mm,185.0mm)
{\fontsize{13}{15}\selectfont $\sqrt{n^2m}=n\sqrt{m}$;\quad divide every term of the numerator}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,193.5mm)
{\fontsize{13}{16}\selectfont $\sqrt{12}=2\sqrt3$,\quad $\dfrac{-4\pm2\sqrt3}{2}=-2\pm\sqrt3$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,206.0mm)
{\fontsize{13}{16}\selectfont \color{accent}$x=-2+\sqrt3$ \ or \ $x=-2-\sqrt3$}
\end{textblock*}
