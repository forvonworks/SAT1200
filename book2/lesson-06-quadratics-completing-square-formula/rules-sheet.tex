% ---- CARD 1: Vertex form ----
\cardbg{16}{40}{178}{58}
\pill{26}{43}{Vertex form (completing the square)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{15}{18}\selectfont $a(x-h)^2+k$, \quad vertex $=(h,k)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,76mm)
{\fontsize{13}{16}\selectfont
$y=x^2-6x+5\ \Longrightarrow\ y=(x-3)^2-4$, \ vertex $(3,-4)$}
\end{textblock*}

% ---- CARD 2: Quadratic formula ----
\cardbg{16}{106}{178}{58}
\pill{26}{109}{Quadratic formula}

\begin{textblock*}{158mm}(26mm,124mm)
{\fontsize{17}{20}\selectfont $x=\dfrac{-b\pm\sqrt{b^2-4ac}}{2a}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,150mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Derived from completing the square on $ax^2+bx+c=0$. Use when factoring
isn't clean.}
\end{textblock*}

% ---- CARD 3: Applying the formula ----
\cardbg{16}{172}{178}{68}
\pill{26}{175}{Applying the formula -- write both roots}

\begin{textblock*}{158mm}(26mm,189mm)
{\fontsize{12}{15}\selectfont
$x^2+4x+1=0$: \ $a=1,b=4,c=1\ \Longrightarrow\ x=\dfrac{-4\pm\sqrt{12}}{2}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,206mm)
{\fontsize{13}{16}\selectfont\color{accent}
$x=-2+\sqrt3$ \ or \ $x=-2-\sqrt3$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,219mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
A quadratic has up to two roots -- write both unless the question asks for
only one.}
\end{textblock*}
