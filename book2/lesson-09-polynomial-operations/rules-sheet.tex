% ---- Degree (y=40 h=34, end 74) ----
\cardbg{16}{40}{178}{34}
\pill{26}{43}{Degree \& leading coefficient (daraja)}

\begin{textblock*}{158mm}(26mm,53.0mm)
{\fontsize{14}{17}\selectfont degree $=$ highest exponent;\quad leading coefficient $=$ its coefficient}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,61.5mm)
{\fontsize{13}{16}\selectfont $5x^3-2x+1$: \ degree $3$, leading coefficient $5$}
\end{textblock*}

% ---- Exponent rules (y=81 h=39, end 120) ----
\cardbg{16}{81}{178}{39}
\pill{26}{84}{Exponent rules}

\begin{textblock*}{158mm}(26mm,94.0mm)
{\fontsize{14}{17}\selectfont $x^m\cdot x^n=x^{m+n}$\quad $(x^m)^n=x^{mn}$\quad $\dfrac{x^m}{x^n}=x^{m-n}$\quad $(ab)^n=a^nb^n$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,106.5mm)
{\fontsize{13}{16}\selectfont $3x^2\cdot4x^3=12x^5$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,112.5mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape Adding keeps the exponent: $x^2+x^2=2x^2$.}
\end{textblock*}

% ---- Add / subtract (y=127 h=39, end 166) ----
\cardbg{16}{127}{178}{39}
\pill{26}{130}{Add / subtract (ko'phad)}

\begin{textblock*}{158mm}(26mm,140.0mm)
{\fontsize{15}{18}\selectfont combine like terms;\quad $A-(B+C)=A-B-C$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,148.5mm)
{\fontsize{13}{16}\selectfont $(5x^3-2x+1)-(2x^3+3x-4)=3x^3-5x+5$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,155.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape Flip every sign in the second polynomial.}
\end{textblock*}

% ---- Multiply (y=173 h=39, end 212) ----
\cardbg{16}{173}{178}{39}
\pill{26}{176}{Multiply polynomials}

\begin{textblock*}{158mm}(26mm,186.0mm)
{\fontsize{15}{18}\selectfont $(a+b)(c+d)=ac+ad+bc+bd$ \quad (every term $\times$ every term)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,194.5mm)
{\fontsize{13}{16}\selectfont $(x+4)(2x^2-x+3)=2x^3+7x^2-x+12$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,201.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape Skip no cross term.}
\end{textblock*}

% ---- Special products (y=219 h=48, end 267) ----
\cardbg{16}{219}{178}{48}
\pill{26}{222}{Special products}

\begin{textblock*}{158mm}(26mm,232.0mm)
{\fontsize{15}{18}\selectfont $(a+b)^2=a^2+2ab+b^2$\qquad $(a-b)^2=a^2-2ab+b^2$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,240.0mm)
{\fontsize{15}{18}\selectfont $(a+b)(a-b)=a^2-b^2$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,249.0mm)
{\fontsize{13}{16}\selectfont $(3x-2)^2=9x^2-12x+4$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,256.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape $(a+b)^2\neq a^2+b^2$ -- the middle term $2ab$ is required.}
\end{textblock*}

\pill{26}{36}{Divide by a monomial}

\begin{textblock*}{158mm}(26mm,46.0mm)

\end{textblock*}

\begin{textblock*}{158mm}(26mm,62.0mm)

% ---- Divide by binomial (y=86 h=39, end 125) ----
\cardbg{16}{86}{178}{39}
\pill{26}{89}{Divide by a binomial (long division)}

\begin{textblock*}{158mm}(26mm,99.0mm)
{\fontsize{15}{18}\selectfont dividend $=$ divisor $\times$ quotient $+$ remainder}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,107.5mm)
{\fontsize{13}{16}\selectfont $(x^2+5x+6)\div(x+2)=x+3$, \ remainder $0$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,114.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape Remainder $0$ means the divisor is a factor.}
\end{textblock*}
