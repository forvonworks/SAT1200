% ---- CARD 1: Add / subtract ----
\cardbg{16}{40}{178}{62}
\pill{26}{43}{Add / subtract polynomials (ko'phad)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{11}{14}\selectfont\color{ink}
Combine like terms. Subtracting: flip every sign in the second
polynomial -- not just the first.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,75mm)
{\fontsize{12.5}{15}\selectfont
$(5x^3-2x+1)-(2x^3+3x-4)=3x^3-5x+5$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,90mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
$x^2+x^2=2x^2$ -- exponents don't change when adding, only when multiplying.}
\end{textblock*}

% ---- CARD 2: Multiply ----
\cardbg{16}{108}{178}{50}
\pill{26}{111}{Multiplying polynomials}

\begin{textblock*}{158mm}(26mm,124mm)
{\fontsize{11}{14}\selectfont\color{ink}
Every term in the first factor times every term in the second, then
combine.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,143mm)
{\fontsize{12.5}{15}\selectfont
$(x+4)(2x^2-x+3)=2x^3+7x^2-x+12$}
\end{textblock*}

% ---- CARD 3: Divide ----
\cardbg{16}{164}{178}{50}
\pill{26}{167}{Dividing polynomials}

\begin{textblock*}{158mm}(26mm,180mm)
{\fontsize{11}{14}\selectfont\color{ink}
By a monomial: divide each term separately. By a binomial: long division,
like numbers.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,199mm)
{\fontsize{12.5}{15}\selectfont
$(x^2+5x+6)\div(x+2)=x+3$ \quad (remainder $0$)}
\end{textblock*}
