\pill{26}{43}{Domain: excluded values}

\begin{textblock*}{158mm}(26mm,55.0mm)
{\fontsize{15}{18}\selectfont $\dfrac{P}{Q}$ is defined only when $Q\neq0$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,70.0mm)
{\fontsize{13}{16}\selectfont $\dfrac{x+1}{x^2-4}$: \ $x\neq2$ and $x\neq-2$}
\end{textblock*}

% ---- Simplify (y=87 h=52, end 139) ----
\cardbg{16}{89}{178}{52}
\pill{26}{92}{Simplify (ratsional ifoda)}

\begin{textblock*}{158mm}(26mm,102.0mm)
{\fontsize{15}{18}\selectfont factor top and bottom, then cancel common \textbf{factors}}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,111.0mm)
{\fontsize{13}{22}\selectfont $\dfrac{x^2-4}{x^2+5x+6}=\dfrac{(x-2)(x+2)}{(x+2)(x+3)}=\dfrac{x-2}{x+3}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,128.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape Never cancel terms: $\dfrac{x+5}{x}\neq5$.}
\end{textblock*}

% ---- Multiply (y=146 h=42, end 188) ----
\cardbg{16}{148}{178}{44}
\pill{26}{151}{Multiply}

\begin{textblock*}{158mm}(26mm,162.0mm)
{\fontsize{15}{18}\selectfont $\dfrac{a}{b}\cdot\dfrac{c}{d}=\dfrac{ac}{bd}$\qquad (factor first, then cancel)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,175.5mm)
{\fontsize{13}{20}\selectfont $\dfrac{x^2-9}{2x}\cdot\dfrac{x}{x+3}=\dfrac{x-3}{2}$}
\end{textblock*}

% ---- Divide (y=195 h=42, end 237) ----
\cardbg{16}{199}{178}{44}
\pill{26}{202}{Divide}

\begin{textblock*}{158mm}(26mm,212.0mm)
{\fontsize{15}{18}\selectfont $\dfrac{a}{b}\div\dfrac{c}{d}=\dfrac{a}{b}\cdot\dfrac{d}{c}$\qquad (multiply by the reciprocal)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,225.5mm)
{\fontsize{13}{20}\selectfont $\dfrac{x^2-16}{x+3}\div\dfrac{x-4}{x+3}=x+4$}
\end{textblock*}

% ---- Add / subtract (y=33 h=62, end 95) ----
\cardbg{16}{33}{178}{56}
\pill{26}{36}{Add / subtract}

\begin{textblock*}{158mm}(26mm,46.0mm)
{\fontsize{15}{18}\selectfont $\dfrac{a}{c}\pm\dfrac{b}{c}=\dfrac{a\pm b}{c}$\qquad $\dfrac{a}{b}+\dfrac{c}{d}=\dfrac{ad+bc}{bd}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,62.0mm)
{\fontsize{13}{20}\selectfont $\dfrac{1}{x}+\dfrac{2}{x+1}=\dfrac{(x+1)+2x}{x(x+1)}=\dfrac{3x+1}{x(x+1)}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,80.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape Get a common denominator first; subtracting flips every sign in the second numerator.}
\end{textblock*}

% ---- Cross-multiplication (y=102 h=46, end 148) ----
\cardbg{16}{96}{178}{44}
\pill{26}{99}{Cross-multiplication}

\begin{textblock*}{158mm}(26mm,110.0mm)
{\fontsize{15}{20}\selectfont $\dfrac{a}{b}=\dfrac{c}{d}\ \Rightarrow\ ad=bc$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,124.0mm)
{\fontsize{13}{20}\selectfont $\dfrac{x}{3}=\dfrac{4}{x+1}\ \Rightarrow\ x(x+1)=12\ \Rightarrow\ x=3$ or $x=-4$}
\end{textblock*}

% ---- Solve (y=155 h=54, end 209) ----
\cardbg{16}{147}{178}{50}
\pill{26}{150}{Solving a rational equation}

\begin{textblock*}{158mm}(26mm,160.0mm)
{\fontsize{14}{17}\selectfont multiply every term by the LCD, solve, then \textbf{check}}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,169.0mm)
{\fontsize{13}{20}\selectfont $\dfrac{x+2}{x-1}=3\ \Rightarrow\ x+2=3(x-1)\ \Rightarrow\ x=\dfrac{5}{2}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,184.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape Check: $x-1=\frac32\neq0$, so $x=\frac52$ is valid.}
\end{textblock*}

% ---- Extraneous (y=216 h=54, end 270) ----
\cardbg{16}{204}{178}{48}
\pill{26}{207}{Extraneous solutions (begona yechim)}

\begin{textblock*}{158mm}(26mm,217.0mm)
{\fontsize{14}{17}\selectfont reject any answer that makes an \textbf{original} denominator $0$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,226.0mm)
{\fontsize{13}{20}\selectfont $\dfrac{x}{x-2}+1=\dfrac{2}{x-2}\ \Rightarrow\ x=2$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,242.0mm)
{\fontsize{13}{16}\selectfont\color{accent} $x-2=0$ at $x=2$: rejected \ $\Rightarrow$ \ \textbf{no solution}}
\end{textblock*}
