% ---- CARD 1: Simplifying rational expressions ----
\cardbg{16}{40}{178}{68}
\pill{26}{43}{Simplifying (ratsional ifoda)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{11}{14}\selectfont\color{ink}
Factor fully, then cancel common \textbf{factors} only -- never bare terms.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,68mm)
{\fontsize{13}{22}\selectfont
$\dfrac{x^2-4}{x^2+5x+6}=\dfrac{(x-2)(x+2)}{(x+2)(x+3)}=\dfrac{x-2}{x+3}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,95mm)
{\fontsize{10.5}{13}\selectfont\color{black!55}\itshape
$x\neq-2,\,-3$ -- values that would make the original denominator $0$ stay
excluded even after canceling.}
\end{textblock*}

% ---- CARD 2: Multiply and divide ----
\cardbg{16}{116}{178}{55}
\pill{26}{119}{Multiplying \& dividing}

\begin{textblock*}{158mm}(26mm,132mm)
{\fontsize{11}{14}\selectfont\color{ink}
Factor first, then cancel. Dividing = multiplying by the reciprocal.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,146mm)
{\fontsize{12}{20}\selectfont
$\dfrac{x^2-9}{2x}\times\dfrac{x}{x+3}=\dfrac{x-3}{2}$}
\end{textblock*}

% ---- CARD 3: Extraneous solutions ----
\cardbg{16}{33}{178}{95}
\pill{26}{36}{Extraneous solutions (begona yechim)}

\begin{textblock*}{158mm}(26mm,49mm)
{\fontsize{11}{14}\selectfont\color{ink}
Clear denominators, solve, then check the candidate in the
\textbf{original} equation -- reject it if any denominator becomes $0$.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,66mm)
{\fontsize{13}{22}\selectfont
$\dfrac{x}{x-2}+1=\dfrac{2}{x-2}\ \Longrightarrow\ x=2$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,92mm)
{\fontsize{12}{15}\selectfont\color{accent}
Check: $x-2=0$ at $x=2$ -- rejected. \textbf{No solution.}}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,108mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Not every candidate is extraneous -- always check, never assume.}
\end{textblock*}
