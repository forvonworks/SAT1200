% ---- CARD 2: Reverse-order trap ----
\cardbg{16}{133}{178}{55}
\pill{26}{136}{``Less than'' reverses the order}

\begin{textblock*}{158mm}(26mm,150mm)
{\fontsize{13}{16}\selectfont
``Eight less than $n$''\ $=\ n-8$ \quad (not $8-n$)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,165mm)
{\fontsize{11}{14}\selectfont
Check: reword as ``[number] minus [amount].'' Same reversal for
``subtracted from.''}
\end{textblock*}
