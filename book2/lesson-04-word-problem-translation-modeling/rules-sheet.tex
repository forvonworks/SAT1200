% ---- CARD 2: Reverse-order trap ----
\cardbg{16}{140}{178}{46}
\pill{26}{143}{``Less than'' reverses the order}

\begin{textblock*}{158mm}(26mm,153mm)
{\fontsize{13}{16}\selectfont
``Eight less than $n$''\ $=\ n-8$ \quad (not $8-n$)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,163mm)
{\fontsize{11}{14}\selectfont
``Six subtracted from $y$''\ $=\ y-6$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,172mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Check: reword as ``[number] minus [amount].''}
\end{textblock*}

% ---- CARD 3: Comparison phrases ----
\cardbg{16}{194}{178}{66}
\pill{26}{197}{Comparison phrases $\to$ inequality}

\begin{textblock*}{158mm}(26mm,207mm)
{\fontsize{10.5}{14}\selectfont
\begin{tabular}{|p{90mm}|>{\centering\arraybackslash}p{48mm}|}
\hline
\rowcolor{accent}\color{white}\textbf{English phrase} & \color{white}\textbf{Symbol} \\ \hline
at least, no less than & $\geq$ \\ \hline
no more than, at most & $\leq$ \\ \hline
more than (comparison) & $>$ \\ \hline
less than, fewer than (comparison) & $<$ \\ \hline
\end{tabular}}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,245mm)
{\fontsize{11}{14}\selectfont
``$x$ increased by 5 is at least 20'' $\Rightarrow x+5\geq 20$ \quad vs. \quad ``$x$ is at least 20'' $\Rightarrow x\geq 20$}
\end{textblock*}

% ---- CARD 4: Total = fixed + rate x quantity ----
\cardbg{16}{33}{178}{34}
\pill{26}{36}{Fixed amount plus rate}

\begin{textblock*}{158mm}(26mm,46mm)
{\fontsize{13}{16}\selectfont
total $=$ fixed $+$ rate $\times$ quantity}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{11}{14}\selectfont
Fee $5$ plus $3$ per item, $4$ items: \ $5+3(4)=17$}
\end{textblock*}

% ---- CARD 5: Distance ----
\cardbg{16}{75}{178}{40}
\pill{26}{78}{Distance, rate, time}

\begin{textblock*}{158mm}(26mm,88mm)
{\fontsize{13}{16}\selectfont
$d=rt$ \hspace{8mm} $r=\dfrac{d}{t}$ \hspace{8mm} $t=\dfrac{d}{r}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,103mm)
{\fontsize{11}{14}\selectfont
$60$ mph for $2.5$ h: \ $d=60(2.5)=150$ miles}
\end{textblock*}

% ---- CARD 6: Consecutive integers ----
\cardbg{16}{123}{178}{38}
\pill{26}{126}{Consecutive integers}

\begin{textblock*}{158mm}(26mm,136mm)
{\fontsize{13}{16}\selectfont
integers: $n,\ n+1,\ n+2$ \hspace{8mm} even / odd: $n,\ n+2,\ n+4$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,146mm)
{\fontsize{11}{14}\selectfont
Three consecutive integers sum to $48$: \ $n+(n+1)+(n+2)=48\ \Rightarrow\ n=15$}
\end{textblock*}

% ---- CARD 7: Ages ----
\cardbg{16}{169}{178}{34}
\pill{26}{172}{Ages}

\begin{textblock*}{158mm}(26mm,182mm)
{\fontsize{13}{16}\selectfont
$t$ years later: $+t$ \hspace{10mm} $t$ years ago: $-t$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,192mm)
{\fontsize{11}{14}\selectfont
Ana is $12$: in $5$ years $12+5=17$; \ $3$ years ago $12-3=9$}
\end{textblock*}

% ---- CARD 8: Percent ----
\cardbg{16}{211}{178}{40}
\pill{26}{214}{Percent of a quantity}

\begin{textblock*}{158mm}(26mm,224mm)
{\fontsize{13}{16}\selectfont
$x\%$ of $y=\dfrac{x}{100}\cdot y$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,239mm)
{\fontsize{11}{14}\selectfont
$15\%$ of $80=0.15\times 80=12$}
\end{textblock*}
