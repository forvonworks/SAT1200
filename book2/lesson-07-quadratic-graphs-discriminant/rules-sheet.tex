\pill{26}{43}{Vertex form (cho'qqi shakli)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{15}{18}\selectfont $y=a(x-h)^2+k$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,72mm)
{\fontsize{11}{14}\selectfont\color{ink}
Vertex $=(h,k)$, axis of symmetry $x=h$. $a>0$ opens up, $a<0$ opens down;
larger $|a|$ = narrower.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,87mm)
{\fontsize{12}{15}\selectfont
$y=3(x-1)^2+4\ \Longrightarrow\ $ vertex $(1,4)$, axis $x=1$, opens up}
\end{textblock*}

% ---- CARD 2: x-intercepts ----
\cardbg{16}{120}{178}{62}
\pill{26}{123}{$x$-intercepts = roots}

\begin{textblock*}{158mm}(26mm,136mm)
{\fontsize{11}{14}\selectfont\color{ink}
Set $y=0$ and solve. ($y$-intercept is different -- the value at $x=0$,
equal to $c$.)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,150mm)
{\fontsize{12}{15}\selectfont
$y=x^2-7x+10\ \Longrightarrow\ (x-2)(x-5)=0\ \Longrightarrow\ x=2$ or $5$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,165mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
``$x$-intercept,'' ``root,'' and ``solution'' all mean the same thing.}
\end{textblock*}

% ---- CARD 3: Discriminant ----
\cardbg{16}{33}{178}{72}
\pill{26}{36}{Discriminant (diskriminant)}

\begin{textblock*}{158mm}(26mm,49mm)
{\fontsize{15}{18}\selectfont $D=b^2-4ac$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,65mm)
{\fontsize{11}{14}\selectfont
$D>0$: 2 real roots \qquad $D=0$: 1 real root \qquad $D<0$: 0 real roots}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,80mm)
{\fontsize{12}{15}\selectfont
$x^2+2x+5=0\ \Longrightarrow\ D=2^2-4(1)(5)=-16\ \Longrightarrow\ $ no real roots}
\end{textblock*}
