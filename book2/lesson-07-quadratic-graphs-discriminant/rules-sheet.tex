\begin{textblock*}{158mm}(26mm,53.0mm)

{\fontsize{13}{16}\selectfont larger $|a|$ $\Rightarrow$ narrower \qquad $y$-intercept $=(0,c)$}

\begin{textblock*}{158mm}(26mm,70.5mm)
{\fontsize{13}{16}\selectfont $y=-2x^2+3x+5$: opens down, $y$-intercept $(0,5)$}

\begin{textblock*}{158mm}(26mm,104.0mm)
{\fontsize{15}{18}\selectfont $y=a(x-h)^2+k$\qquad vertex $(h,k)$, axis $x=h$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,113.5mm)
{\fontsize{13}{16}\selectfont $y=3(x-1)^2+4$: vertex $(1,4)$, axis $x=1$, opens up}

% ---- Vertex from standard form; min / max (y=134 h=50, end 184) ----

\pill{26}{137}{Vertex from $ax^2+bx+c$; min / max}

\begin{textblock*}{158mm}(26mm,147.0mm)
{\fontsize{15}{18}\selectfont $x_{\text{vertex}}=-\dfrac{b}{2a}$,\quad $y_{\text{vertex}}=f\!\left(-\dfrac{b}{2a}\right)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,163.5mm)
{\fontsize{13}{16}\selectfont $a>0$: minimum value $k$ \qquad $a<0$: maximum value $k$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,172.0mm)
{\fontsize{13}{16}\selectfont $y=x^2-6x+5$: $x=3$, $y=-4$ $\Rightarrow$ vertex $(3,-4)$, min $=-4$}
