\pill{26}{43}{Random sampling (tasodifiy tanlanma) $\Rightarrow$ generalize}

\begin{textblock*}{158mm}(26mm,53.0mm)
{\fontsize{13}{16}\selectfont random sample $\Rightarrow$ result applies to the whole population}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,61.0mm)
{\fontsize{13}{16}\selectfont convenience / voluntary-response sample $\Rightarrow$ describes only itself}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,70.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape A bigger sample improves precision, not representativeness.}
\end{textblock*}

% ---- Random assignment (y=85 h=38, end 123) ----
\cardbg{16}{85}{178}{38}
\pill{26}{88}{Random assignment (tasodifiy taqsimlash) $\Rightarrow$ cause-and-effect}

\begin{textblock*}{158mm}(26mm,98.0mm)
{\fontsize{13}{16}\selectfont random assignment to groups $\Rightarrow$ ``X causes Y''}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,106.0mm)
{\fontsize{13}{16}\selectfont observational study (no assignment) $\Rightarrow$ ``X is associated with Y'' only}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,115.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape A lurking variable may drive both X and Y.}
\end{textblock*}

% ---- Decision table (y=130 h=68, end 198) ----
\cardbg{16}{130}{178}{68}
\pill{26}{133}{What can this study prove?}

\begin{textblock*}{158mm}(26mm,143.0mm)
{\fontsize{11}{14}\selectfont
\begin{tabular}{|>{\centering\arraybackslash}p{30mm}|>{\centering\arraybackslash}p{40mm}|p{58mm}|}
\hline
\cellcolor{accent}\textcolor{white}{\textbf{Random sample?}} & \cellcolor{accent}\textcolor{white}{\textbf{Random assignment?}} & \cellcolor{accent}\textcolor{white}{\textbf{Can claim}} \\ \hline
Yes & Yes & cause-and-effect, whole population \\ \hline
Yes & No & association only, whole population \\ \hline
No & Yes & cause-and-effect, this group only \\ \hline
No & No & neither \\ \hline
\end{tabular}}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,187.0mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape Sampling answers ``who?'' (generalize); assignment answers ``which group?'' (cause).}
\end{textblock*}
