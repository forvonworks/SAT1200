% ---- CARD 1: Random sample -> generalize ----
\cardbg{16}{40}{178}{48}
\pill{26}{43}{Random sample (tasodifiy tanlanma) $\to$ generalize}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{11}{14}\selectfont\color{ink}
Random sample $\to$ conclusion applies to the whole population. Convenience
sample $\to$ describes only itself, no matter how large.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,72mm)
{\fontsize{10.5}{13}\selectfont\color{black!55}\itshape
Size improves precision; only randomness improves representativeness.}
\end{textblock*}

% ---- CARD 2: Random assignment -> causation ----
\cardbg{16}{96}{178}{52}
\pill{26}{99}{Random assignment $\to$ causation}

\begin{textblock*}{158mm}(26mm,112mm)
{\fontsize{11}{14}\selectfont\color{ink}
Random assignment to groups $\to$ ``X causes Y.'' Observational study (no
assignment) $\to$ ``X is associated with Y'' only -- never causation.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,131mm)
{\fontsize{10.5}{13}\selectfont\color{black!55}\itshape
A lurking variable can drive both at once -- e.g. ice-cream sales and
drownings both rise with hot weather, not with each other.}
\end{textblock*}
