% ---- CARD 1: Part-to-part vs part-to-whole ----
\cardbg{16}{40}{178}{50}
\pill{26}{43}{Ratio (nisbat): part-to-part vs part-to-whole}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{11}{14}\selectfont\color{ink}
A $2:3$ ratio has $2+3=5$ total parts. Each share $=\dfrac{\text{its
number}}{\text{total parts}}$.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,70mm)
{\fontsize{13}{16}\selectfont
$20$ total, ratio $2:3\ \Longrightarrow\ \dfrac{2}{5}(20)=8$ and $\dfrac{3}{5}(20)=12$}
\end{textblock*}

% ---- CARD 2: Unit rate ----
\cardbg{16}{98}{178}{45}
\pill{26}{101}{Unit rate}

\begin{textblock*}{158mm}(26mm,114mm)
{\fontsize{13}{16}\selectfont
unit rate $=\dfrac{\text{quantity}}{\text{time (or other unit)}}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,128mm)
{\fontsize{11}{14}\selectfont
$180$ km in $3$ h $\ \Longrightarrow\ 180\div3=60$ km/h}
\end{textblock*}

% ---- CARD 3: Solving a proportion ----
\cardbg{16}{151}{178}{80}
\pill{26}{154}{Solving a proportion (proportsiya)}

\begin{textblock*}{158mm}(26mm,167mm)
{\fontsize{15}{18}\selectfont
$\dfrac{a}{b}=\dfrac{c}{d}\ \Longrightarrow\ a\times d=b\times c$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,182mm)
{\fontsize{12}{15}\selectfont
$\dfrac{x}{5}=\dfrac{12}{20}\ \Longrightarrow\ 20x=60\ \Longrightarrow\ x=3$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,194mm)
{\fontsize{12}{15}\selectfont
Flour:sugar $=3:5$, $18$ cups flour $\ \Longrightarrow\ \dfrac{3}{5}=\dfrac{18}{s}\ \Longrightarrow\ s=30$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,209mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Keep the same quantity on top on both sides, and match units before you
set up the proportion.}
\end{textblock*}
