% ---- CARD 1: Ratio ----
\cardbg{16}{40}{178}{38}
\pill{26}{43}{Ratio (nisbat)}

\begin{textblock*}{158mm}(26mm,54mm)
{\fontsize{14}{17}\selectfont
$a:b=\dfrac{a}{b}=\dfrac{ka}{kb}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,67mm)
{\fontsize{11}{14}\selectfont
Simplify by dividing both terms by the same number: \ $12:18=2:3$}
\end{textblock*}

% ---- CARD 2: Part-to-part vs part-to-whole ----
\cardbg{16}{86}{178}{42}
\pill{26}{89}{Part-to-part vs part-to-whole}

\begin{textblock*}{158mm}(26mm,100mm)
{\fontsize{14}{17}\selectfont
$a:b\ \Rightarrow\ \text{whole}=a+b$ parts \hspace{6mm}
share of whole $=\dfrac{a}{a+b}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,115mm)
{\fontsize{11}{14}\selectfont
$20$ total, ratio $2:3$: \ $\dfrac{2}{5}(20)=8$ \ and \ $\dfrac{3}{5}(20)=12$}
\end{textblock*}

% ---- CARD 3: Rate and unit rate ----
\cardbg{16}{136}{178}{40}
\pill{26}{139}{Rate and unit rate (birlik tezlik)}

\begin{textblock*}{158mm}(26mm,150mm)
{\fontsize{14}{17}\selectfont
rate $=\dfrac{\text{quantity}}{\text{time}}$ \hspace{10mm} $d=rt$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,165mm)
{\fontsize{11}{14}\selectfont
$180$ km in $3$ h: \ $180\div3=60$ km/h}
\end{textblock*}

% ---- CARD 4: Solving a proportion ----
\cardbg{16}{184}{178}{46}
\pill{26}{187}{Solving a proportion (proportsiya)}

\begin{textblock*}{158mm}(26mm,198mm)
{\fontsize{15}{18}\selectfont
$\dfrac{a}{b}=\dfrac{c}{d}\ \Longrightarrow\ a\times d=b\times c$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,213mm)
{\fontsize{11}{14}\selectfont
Flour:sugar $=3:5$, $18$ cups flour: \ $\dfrac{3}{5}=\dfrac{18}{s}\ \Longrightarrow\ 3s=90\ \Longrightarrow\ s=30$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,225mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Same quantity on top on both sides; match units first.}
\end{textblock*}

% ---- CARD 5: Scale ----
\cardbg{16}{238}{178}{40}
\pill{26}{241}{Scale (masshtab)}

\begin{textblock*}{158mm}(26mm,252mm)
{\fontsize{14}{17}\selectfont
$\dfrac{\text{map length}}{\text{actual length}}=$ constant}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,267mm)
{\fontsize{11}{14}\selectfont
$1$ in $=15$ mi: \ $\dfrac{1}{15}=\dfrac{4.5}{m}\ \Longrightarrow\ m=67.5$ mi}
\end{textblock*}

% ---- CARD 6: Unit conversion ----
\cardbg{16}{33}{178}{38}
\pill{26}{36}{Unit conversion}

\begin{textblock*}{158mm}(26mm,47mm)
{\fontsize{14}{17}\selectfont
multiply by conversion fractions so the units cancel}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,57mm)
{\fontsize{11}{14}\selectfont
$90\,\dfrac{\text{km}}{\text{h}}\times\dfrac{1000\text{ m}}{1\text{ km}}\times\dfrac{1\text{ h}}{3600\text{ s}}=25\text{ m/s}$}
\end{textblock*}

% ---- CARD 7: Average rate ----
\cardbg{16}{79}{178}{52}
\pill{26}{82}{Average rate}

\begin{textblock*}{158mm}(26mm,93mm)
{\fontsize{14}{17}\selectfont
average rate $=\dfrac{\text{total quantity}}{\text{total time}}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,108mm)
{\fontsize{11}{14}\selectfont
$120$ km at $60$ km/h, then $120$ km at $40$ km/h: \ $\dfrac{240}{2+3}=48$ km/h}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,120mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Not the mean of the two rates ($50$) unless the times are equal.}
\end{textblock*}
