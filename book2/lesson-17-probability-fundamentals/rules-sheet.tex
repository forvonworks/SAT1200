% ---- CARD 1: Basic probability ----
\cardbg{16}{40}{178}{68}
\pill{26}{43}{Basic probability (ehtimollik)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{15}{18}\selectfont $P(\text{event})=\dfrac{\text{favorable outcomes}}{\text{total outcomes}}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,78mm)
{\fontsize{11}{14}\selectfont
200 students, 80 prefer basketball: $P=\dfrac{80}{200}=\dfrac25=40\%$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,90mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
From a table: total outcomes is the relevant row/column total -- read the
condition carefully before dividing.}
\end{textblock*}

% ---- CARD 2: Addition rule ----
\cardbg{16}{116}{178}{78}
\pill{26}{119}{Addition rule: $P(A$ or $B)$}

\begin{textblock*}{158mm}(26mm,132mm)
{\fontsize{13}{16}\selectfont $P(A\text{ or }B)=P(A)+P(B)-P(A\text{ and }B)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,147mm)
{\fontsize{11}{14}\selectfont\color{ink}
One die, even OR $>3$: $\dfrac36+\dfrac36-\dfrac26=\dfrac23$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,162mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
If $A$ and $B$ can't happen together, $P(A\text{ and }B)=0$ and the rule
reduces to $P(A)+P(B)$ -- but always check for overlap first.}
\end{textblock*}

% ---- CARD 3: Multiplication rule ----
\cardbg{16}{33}{178}{58}
\pill{26}{36}{Multiplication rule: independent events}

\begin{textblock*}{158mm}(26mm,49mm)
{\fontsize{13}{16}\selectfont $P(A\text{ and }B)=P(A)\times P(B)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,64mm)
{\fontsize{11}{14}\selectfont\color{ink}
Two coin flips, heads then tails: $\dfrac12\times\dfrac12=\dfrac14$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,78mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Only valid when one event's outcome doesn't change the other's probability.}
\end{textblock*}

% ---- CARD 4: Mutually exclusive vs independent ----
\cardbg{16}{101}{178}{115}
\pill{26}{104}{Mutually exclusive vs.\ independent}

\begin{textblock*}{158mm}(26mm,117mm)
{\fontsize{10.5}{14}\selectfont
\begin{tabular}{|p{32mm}|p{52mm}|p{52mm}|}
\hline
\rowcolor{accent}
{\color{white}\textbf{}} & {\color{white}\textbf{Mutually exclusive}} & {\color{white}\textbf{Independent}} \\ \hline
Question & Can they happen together at all? & Does one change the other's probability? \\ \hline
Condition & $P(A\text{ and }B)=0$ & $P(A\text{ and }B)=P(A)P(B)$ \\ \hline
Rule used & Addition rule & Multiplication rule \\ \hline
\end{tabular}}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,172mm)
{\fontsize{11}{14}\selectfont\color{accent}
If two events are mutually exclusive AND both have nonzero probability,
they are automatically NOT independent.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,192mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
One die: "rolling 3" and "rolling 4" are mutually exclusive but not
independent -- rolling a 3 makes $P(4)$ drop from $\frac16$ to $0$.}
\end{textblock*}
