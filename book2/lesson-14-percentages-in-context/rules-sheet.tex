% ---- CARD 1: Percent of a quantity ----
\cardbg{16}{40}{178}{58}
\pill{26}{43}{Part, percent, whole}

\begin{textblock*}{158mm}(26mm,54mm)
{\fontsize{14}{17}\selectfont
part $=$ percent $\times$ whole \hspace{8mm} percent $=\dfrac{\text{part}}{\text{whole}}\times100$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,69mm)
{\fontsize{14}{17}\selectfont
whole $=\dfrac{\text{part}}{\text{percent}}$ \hspace{8mm} (use the decimal form: $15\%=0.15$)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,86mm)
{\fontsize{11}{14}\selectfont
$15\%$ of $80=0.15\times80=12$}
\end{textblock*}

% ---- CARD 2: Percent change ----
\cardbg{16}{106}{178}{50}
\pill{26}{109}{Percent change (foiz o'zgarishi)}

\begin{textblock*}{158mm}(26mm,120mm)
{\fontsize{15}{18}\selectfont
percent change $=\dfrac{\text{new}-\text{old}}{\text{old}}\times100\%$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,135mm)
{\fontsize{11}{14}\selectfont
$\$40\to\$52$: \ $\dfrac{52-40}{40}=0.30=30\%$ increase}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,146mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Always divide by the old value.}
\end{textblock*}

% ---- CARD 3: Multiplier method ----
\cardbg{16}{164}{178}{38}
\pill{26}{167}{Multiplier method (ko'paytuvchi usuli)}

\begin{textblock*}{158mm}(26mm,178mm)
{\fontsize{14}{17}\selectfont
new $=$ old $\times(1\pm r)$ \hspace{8mm} $+$ increase, $-$ decrease}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,189mm)
{\fontsize{11}{14}\selectfont
$\$150$, $+20\%$: \ $150\times1.20=\$180$}
\end{textblock*}

% ---- CARD 4: Original from final ----
\cardbg{16}{210}{178}{42}
\pill{26}{213}{Original from the final value}

\begin{textblock*}{158mm}(26mm,224mm)
{\fontsize{14}{17}\selectfont
old $=\dfrac{\text{new}}{1\pm r}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,238mm)
{\fontsize{11}{14}\selectfont
After a $20\%$ discount the price is $\$60$: \ $\dfrac{60}{1-0.20}=\$75$}
\end{textblock*}

% ---- CARD 5: Successive percent changes ----
\cardbg{16}{33}{178}{58}
\pill{26}{36}{Successive percent changes}

\begin{textblock*}{158mm}(26mm,47mm)
{\fontsize{14}{17}\selectfont
final $=$ old $\times(1\pm r_1)\times(1\pm r_2)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,58mm)
{\fontsize{14}{17}\selectfont
net change $=(\text{product of factors}-1)\times100\%$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,70mm)
{\fontsize{11}{14}\selectfont
$\$500$, $+20\%$ then $-10\%$: \ $500\times1.20\times0.90=\$540$, net $+8\%$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,78mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Percent changes multiply; they never add or subtract.}
\end{textblock*}

% ---- CARD 6: Discount, markup, tax, tip ----
\cardbg{16}{99}{178}{40}
\pill{26}{102}{Discount, markup, tax, tip}

\begin{textblock*}{158mm}(26mm,113mm)
{\fontsize{14}{17}\selectfont
discount $d$: $\times(1-d)$ \hspace{8mm} markup, tax, tip $r$: $\times(1+r)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,125mm)
{\fontsize{11}{14}\selectfont
$\$80$, $25\%$ off, then $10\%$ tax: \ $80\times0.75\times1.10=\$66$}
\end{textblock*}
