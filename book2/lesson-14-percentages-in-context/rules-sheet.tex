% ---- CARD 1: Percent change ----
\cardbg{16}{40}{178}{65}
\pill{26}{43}{Percent change (foiz o'zgarishi)}

\begin{textblock*}{158mm}(26mm,58mm)
{\fontsize{15}{20}\selectfont $\dfrac{N-O}{O}\times100\%$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,80mm)
{\fontsize{11}{14}\selectfont\color{ink}
$N$ = new value, $O$ = original value -- always divide by the
\textbf{original}, never the new value.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,94mm)
{\fontsize{12}{15}\selectfont
$\$40\to\$52$: $\ \dfrac{52-40}{40}=30\%$ increase}
\end{textblock*}

% ---- CARD 2: Multiplier method ----
\cardbg{16}{115}{178}{55}
\pill{26}{118}{Multiplier method (ko'paytuvchi usuli)}

\begin{textblock*}{158mm}(26mm,133mm)
{\fontsize{15}{20}\selectfont $N=O\times(1\pm r)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,150mm)
{\fontsize{11}{14}\selectfont\color{ink}
$r$ = percent as a decimal. $+$ for increase, $-$ for decrease.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,160mm)
{\fontsize{12}{15}\selectfont
$\$150$, $+20\%$: $\ 150\times1.20=\$180$}
\end{textblock*}

% ---- CARD 3: Successive percent changes ----
\cardbg{16}{180}{178}{80}
\pill{26}{183}{Successive percent changes}

\begin{textblock*}{158mm}(26mm,198mm)
{\fontsize{11}{14}\selectfont\color{ink}
Each change applies to the \textbf{previous result}, not the original.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,209mm)
{\fontsize{15}{20}\selectfont $O\times(1+r_1)\times(1-r_2)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,226mm)
{\fontsize{12}{15}\selectfont
$\$500$, $+20\%$ then $-10\%$: $\ 500\times1.20\times0.90=\$540$ (net $+8\%$,
not $+10\%$)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,245mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Percent changes never add or subtract directly -- multiply the factors.}
\end{textblock*}
