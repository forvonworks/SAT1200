% ---- CARD 1: Linear vs exponential (table test) ----
\cardbg{16}{40}{178}{75}
\pill{26}{43}{Linear vs. exponential (table test)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
Common DIFFERENCE constant $\to$ linear. Common RATIO constant $\to$
exponential.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,68mm)
{\fontsize{10.5}{13}\selectfont
\begin{tabular}{|p{28mm}|p{60mm}|>{\centering\arraybackslash}p{54mm}|}
\hline
\rowcolor{accent}
{\color{white}\textbf{Type}} & {\color{white}\textbf{Values}} & {\color{white}\textbf{Test}} \\ \hline
Linear & $3,\ 7,\ 11,\ 15$ & difference always $+4$ \\ \hline
Exponential & $5,\ 10,\ 20,\ 40$ & ratio always $\times2$ \\ \hline
\end{tabular}}
\end{textblock*}

% ---- CARD 2: Growth/decay formula ----
\cardbg{16}{124}{178}{70}
\pill{26}{127}{Growth / decay formula}

\begin{textblock*}{158mm}(26mm,140mm)
{\fontsize{14}{17}\selectfont
$y=a(1+r)^x$ (growth) \qquad $y=a(1-r)^x$ (decay)}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,158mm)
{\fontsize{11}{14}\selectfont
$a=200$, $r=5\%$, $x=4$ yr: \quad $y=200(1.05)^4\approx243$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,172mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
$r$ is a decimal: $7\%=0.07$, not $1.7$.}
\end{textblock*}

% ---- CARD 3: Half-life ----
\cardbg{16}{203}{178}{65}
\pill{26}{206}{Half-life}

\begin{textblock*}{158mm}(26mm,219mm)
{\fontsize{14}{17}\selectfont $y=a\left(\dfrac12\right)^{t/h}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,240mm)
{\fontsize{11}{14}\selectfont
$a=160$ g, $h=4$ yr, $t=12$ yr: \quad $y=160\left(\dfrac12\right)^{12/4}=160\left(\dfrac12\right)^3=20$ g}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,257mm)
{\fontsize{10}{13}\selectfont\color{black!55}\itshape
Each half-life multiplies the remaining amount by $\frac12$ -- it does not
subtract a fixed amount.}
\end{textblock*}
