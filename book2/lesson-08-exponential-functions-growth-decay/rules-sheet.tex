% ---- Exponential function (y=40 h=44, end 84) ----

\pill{26}{43}{Exponential function $y=a\cdot b^x$}

\begin{textblock*}{158mm}(26mm,53.0mm)
{\fontsize{15}{18}\selectfont $a$ = initial value (boshlang'ich qiymat), passes through $(0,a)$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,61.5mm)
{\fontsize{13}{16}\selectfont $b$ = growth factor: \ $b>1$ growth \qquad $0<b<1$ decay}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,70.5mm)
{\fontsize{13}{16}\selectfont $y=3\cdot2^x$: initial value $3$, factor $2$, passes through $(0,3)$}
\end{textblock*}

% ---- Linear vs exponential (y=91 h=52, end 143) ----
\cardbg{16}{91}{178}{52}
\pill{26}{94}{Linear vs.\ exponential}

\begin{textblock*}{158mm}(26mm,104.0mm)
{\fontsize{13}{16}\selectfont constant difference $\Rightarrow$ $y=mx+b$ \qquad constant ratio $\Rightarrow$ $y=a\cdot b^x$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,113.0mm)
{\fontsize{11}{14}\selectfont
\begin{tabular}{|p{24mm}|p{36mm}|p{36mm}|p{30mm}|}
\hline
\cellcolor{accent}\textcolor{white}{\textbf{Type}} & \cellcolor{accent}\textcolor{white}{\textbf{$y$ at $x=0,1,2,3$}} & \cellcolor{accent}\textcolor{white}{\textbf{Test}} & \cellcolor{accent}\textcolor{white}{\textbf{Rule}} \\ \hline
Linear & $3,\ 7,\ 11,\ 15$ & difference $+4$ & $y=4x+3$ \\ \hline
Exponential & $5,\ 10,\ 20,\ 40$ & ratio $\times2$ & $y=5\cdot2^x$ \\ \hline
\end{tabular}}
\end{textblock*}

% ---- Percent rate to factor (y=150 h=42, end 192) ----
\cardbg{16}{150}{178}{42}
\pill{26}{153}{Percent rate $\to$ factor}

\begin{textblock*}{158mm}(26mm,163.0mm)
{\fontsize{15}{18}\selectfont $y=a(1+r)^t$ growth \qquad $y=a(1-r)^t$ decay}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,171.5mm)
{\fontsize{13}{16}\selectfont $r$ as a decimal: $5\%$ growth $\to 1.05$ \qquad $12\%$ decay $\to 0.88$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,180.0mm)
{\fontsize{13}{16}\selectfont $a=200$, $r=5\%$, $t=4$: $\ y=200(1.05)^4\approx243.1$}
\end{textblock*}

% ---- Change every k units (y=33 h=50, end 85) ----
\cardbg{16}{33}{178}{52}
\pill{26}{36}{Change every $k$ units}

\begin{textblock*}{158mm}(26mm,46.0mm)
{\fontsize{15}{18}\selectfont $y=a\cdot b^{t/k}$ \qquad $b$ = factor applied once every $k$ units of $t$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,55.0mm)
{\fontsize{15}{18}\selectfont doubling: $a\cdot2^{t/d}$ \qquad half-life: $a\left(\dfrac12\right)^{t/h}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,69.0mm)
{\fontsize{13}{16}\selectfont $a=160$ g, $h=4$ yr, $t=12$ yr: $\ y=160\left(\dfrac12\right)^{12/4}=160\left(\dfrac12\right)^3=20$ g}
\end{textblock*}

% ---- Compound interest (y=92 h=48, end 140) ----
\cardbg{16}{92}{178}{48}
\pill{26}{95}{Compound interest}

\begin{textblock*}{158mm}(26mm,105.0mm)
{\fontsize{15}{18}\selectfont $A=P\left(1+\dfrac{r}{n}\right)^{nt}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,120.0mm)
{\fontsize{13}{16}\selectfont $P$ principal, $r$ annual rate (decimal), $n$ times per year, $t$ years}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,128.0mm)
{\fontsize{13}{16}\selectfont $P=1000$, $r=6\%$, $n=4$, $t=2$: $\ A=1000(1.015)^8\approx1126.49$}
\end{textblock*}
