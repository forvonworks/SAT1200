% ---- CARD 1: Mean, median, mode ----
\cardbg{16}{40}{178}{65}
\pill{26}{43}{Mean, median, mode (o'rta qiymatlar)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{10.5}{14}\selectfont
\begin{tabular}{|p{30mm}|p{118mm}|}
\hline
\rowcolor{accent}
{\color{white}\textbf{Measure}} & {\color{white}\textbf{Rule}} \\ \hline
Mean & sum $\div$ count \\ \hline
Median & middle value, sorted -- odd count: middle; even count: average of
the two middle values \\ \hline
Mode & most frequent value \\ \hline
\end{tabular}}
\end{textblock*}

% ---- CARD 2: Weighted mean ----
\cardbg{16}{113}{178}{60}
\pill{26}{116}{Weighted mean (combining groups)}

\begin{textblock*}{158mm}(26mm,129mm)
{\fontsize{13}{16}\selectfont
Combined mean $=\dfrac{\text{sum}_A+\text{sum}_B}{\text{count}_A+\text{count}_B}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,148mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
Not $\dfrac{\text{mean}_A+\text{mean}_B}{2}$ -- unless the two groups are
the same size.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,160mm)
{\fontsize{11}{14}\selectfont
$8$ students, mean $70$; $12$ students, mean $90$:
$\dfrac{8(70)+12(90)}{8+12}=\dfrac{1640}{20}=82$}
\end{textblock*}

% ---- CARD 3: Range and outlier sensitivity ----
\cardbg{16}{181}{178}{65}
\pill{26}{184}{Range \& outlier sensitivity}

\begin{textblock*}{158mm}(26mm,197mm)
{\fontsize{15}{18}\selectfont $\text{range}=\text{max}-\text{min}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,213mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
An outlier pulls the mean strongly; the median barely moves -- it depends
only on sorted position, not size.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,228mm)
{\fontsize{11}{14}\selectfont
$\{4,5,6,7,8\}$, mean $=6$, median $=6$. Add $50$: mean
$\approx13.33$, median $=6.5$.}
\end{textblock*}

% ---- CARD 4: Standard deviation from the graph ----
\cardbg{16}{33}{178}{91}
\pill{26}{36}{Standard deviation (SD) -- read it from the graph}

\begin{textblock*}{158mm}(26mm,49mm)
{\fontsize{12}{15}\selectfont
SD $=$ how spread out the data is around the mean.\\
\textbf{More spread $\Rightarrow$ bigger SD.} No formula needed.}
\end{textblock*}

\begin{textblock*}{50mm}(26mm,65mm)
{\sffamily\bfseries\fontsize{9}{10}\selectfont Skyrocket: \textcolor{accent}{small SD}}
\end{textblock*}
\begin{textblock*}{50mm}(78mm,65mm)
{\sffamily\bfseries\fontsize{9}{10}\selectfont Double top: \textcolor{accent}{medium SD}}
\end{textblock*}
\begin{textblock*}{50mm}(130mm,65mm)
{\sffamily\bfseries\fontsize{9}{10}\selectfont Spread out: \textcolor{accent}{large SD}}
\end{textblock*}

\begin{textblock*}{48mm}(26mm,71mm)
\bargraph{0,0,0,3,14,3,0,0,0}{4.4}{1.7}{9}{34}{5}{15.5}
\end{textblock*}
\begin{textblock*}{48mm}(78mm,71mm)
\bargraph{0,0,8,0,4,0,8,0,0}{4.4}{1.7}{9}{34}{5}{15.5}
\end{textblock*}
\begin{textblock*}{48mm}(130mm,71mm)
\bargraph{2,2,2,2,4,2,2,2,2}{4.4}{1.7}{9}{34}{5}{15.5}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,108mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
Same 20 values and same mean $(5)$ in all three -- only the spread differs.\\
Move the two peaks to the far ends ($1$ and $9$) $\Rightarrow$ even bigger SD.}
\end{textblock*}

% ---- CARD 6: Spread <-> SD ----
\cardbg{16}{33}{178}{70}
\pill{26}{36}{Spread and SD move together}

\begin{textblock*}{158mm}(26mm,49mm)
{\fontsize{10.5}{13}\selectfont
\begin{tabular}{|p{82mm}|p{60mm}|}
\hline
\cellcolor{accent}\textcolor{white}{\textbf{What happens to the data}} & \cellcolor{accent}\textcolor{white}{\textbf{SD}} \\ \hline
Values closer together & gets smaller \\ \hline
Values farther apart & gets bigger \\ \hline
All values equal & $=0$ (never negative) \\ \hline
Add or subtract the same number to every value & stays the same \\ \hline
Multiply every value by $k$ & multiplied by $|k|$ \\ \hline
Add one far-away outlier & jumps up \\ \hline
\end{tabular}}
\end{textblock*}

% ---- CARD 7: Which set has the larger SD? ----
\cardbg{16}{111}{178}{52}
\pill{26}{114}{Which set has the larger SD?}

\begin{textblock*}{158mm}(26mm,127mm)
{\fontsize{12}{15}\selectfont
Compare how far values sit from \emph{their own} mean --\\
not how large the values are.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,144mm)
{\fontsize{11}{14}\selectfont
$\{48,49,50,51,52\}$ vs $\{10,30,50,70,90\}$: same mean $50$, second is far
more spread $\Rightarrow$ larger SD.\\[1mm]
$\{101,102,103\}$ and $\{1,2,3\}$: same spread $\Rightarrow$ same SD.}
\end{textblock*}
