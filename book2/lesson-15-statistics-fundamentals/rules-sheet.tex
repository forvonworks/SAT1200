% ---- CARD 1: Mean, median, mode ----
\cardbg{16}{40}{178}{65}
\pill{26}{43}{Mean, median, mode (o'rta qiymatlar)}

\begin{textblock*}{158mm}(26mm,56mm)
{\fontsize{10.5}{14}\selectfont
\begin{tabular}{|p{30mm}|p{118mm}|}
\hline
\rowcolor{accent}
{\color{white}\textbf{Measure}} & {\color{white}\textbf{Rule}} \\ \hline
Mean & sum $\div$ count \\ \hline
Median & middle value, sorted -- odd count: middle; even count: average of
the two middle values \\ \hline
Mode & most frequent value \\ \hline
\end{tabular}}
\end{textblock*}

% ---- CARD 2: Weighted mean ----
\cardbg{16}{113}{178}{60}
\pill{26}{116}{Weighted mean (combining groups)}

\begin{textblock*}{158mm}(26mm,129mm)
{\fontsize{13}{16}\selectfont
Combined mean $=\dfrac{\text{sum}_A+\text{sum}_B}{\text{count}_A+\text{count}_B}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,148mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
Not $\dfrac{\text{mean}_A+\text{mean}_B}{2}$ -- unless the two groups are
the same size.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,160mm)
{\fontsize{11}{14}\selectfont
$8$ students, mean $70$; $12$ students, mean $90$:
$\dfrac{8(70)+12(90)}{8+12}=\dfrac{1640}{20}=82$}
\end{textblock*}

% ---- CARD 3: Range and outlier sensitivity ----
\cardbg{16}{181}{178}{65}
\pill{26}{184}{Range \& outlier sensitivity}

\begin{textblock*}{158mm}(26mm,197mm)
{\fontsize{15}{18}\selectfont $\text{range}=\text{max}-\text{min}$}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,213mm)
{\fontsize{10.5}{13}\selectfont\color{ink}
An outlier pulls the mean strongly; the median barely moves -- it depends
only on sorted position, not size.}
\end{textblock*}

\begin{textblock*}{158mm}(26mm,228mm)
{\fontsize{11}{14}\selectfont
$\{4,5,6,7,8\}$, mean $=6$, median $=6$. Add $50$: mean
$\approx13.33$, median $=6.5$.}
\end{textblock*}
